# 反射式记忆管理（RMM）系统技术方案
## 一、系统初始化与组件配置
### 1. 记忆库初始化
**步骤**：创建一个空的记忆库 B，用于存储所有记忆条目。每个记忆条目采用结构化格式：

```
(topic_summary, raw_dialogue_segment)
```
其中 `topic_summary`作为检索键。

**技术选型**：记忆库基于向量数据库实现，以支持高效的相似性搜索。

**代码实现**：

```
class MemoryBank:
    def __init__(self, vector_db_config):
        self.vector_db = VectorDatabase(**vector_db_config)
        self.memories = []  # 存储完整记忆条目
        
    def add_memory(self, topic_summary, raw_dialogue, embedding):
        """添加新记忆条目"""
        memory_id = len(self.memories)
        memory_entry = {
            'id': memory_id,
            'topic_summary': topic_summary,
            'raw_dialogue': raw_dialogue,
            'embedding': embedding
        }
        self.memories.append(memory_entry)
        self.vector_db.add(embedding, memory_id)
        
    def search(self, query_embedding, top_k=10):
        """相似性搜索"""
        return self.vector_db.search(query_embedding, top_k)
```

import torch
import torch.nn as nn

class DenseRetriever:
    def __init__(self, model_name="sentence-transformers/all-mpnet-base-v2"):
        self.model = AutoModel.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
    
    def encode(self, text):
        """将文本编码为向量"""
        inputs = self.tokenizer(text, return_tensors='pt', padding=True, truncation=True)
        with torch.no_grad():
            outputs = self.model(**inputs)
        return outputs.last_hidden_state.mean(dim=1).squeeze()

class Reranker(nn.Module):
    def __init__(self, hidden_size=768):
        super().__init__()
        self.linear1 = nn.Linear(hidden_size * 3, hidden_size)  # 查询+记忆+交互特征
        self.linear2 = nn.Linear(hidden_size, 1)
        self.dropout = nn.Dropout(0.1)
    
    def forward(self, query_emb, memory_emb, interaction_features):
        x = torch.cat([query_emb, memory_emb, interaction_features], dim=-1)
        x = torch.relu(self.linear1(x))
        x = self.dropout(x)
        return torch.sigmoid(self.linear2(x))

# 初始化组件
retriever = DenseRetriever()
reranker = Reranker()
```

class DialogueGenerator:
    def __init__(self, model_name, generation_config=None):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.generation_config = generation_config or {
            'max_length': 512,
            'temperature': 0.7,
            'do_sample': True
        }
    
    def generate(self, prompt):
        """生成回复"""
        inputs = self.tokenizer(prompt, return_tensors='pt')
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                **self.generation_config
            )
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)
```
## 二、对话会话中的在线处理流程
### 算法1：单次查询处理流程
```
def online_processing_algorithm(q, B, S, f_θ, g_φ, generator):
    """
    处理单次用户查询的完整流程
    """
    # 1. 记忆检索
    M_K = memory_retrieval(q, B, f_θ, top_k=10)
    
    # 2. 记忆重排序
    M_M = memory_reranking(q, M_K, g_φ, top_m=5)
    
    # 3. 响应生成与归因分析
    a, R_M = response_generation_with_attribution(q, S, M_M, generator)
    
    # 4. 回溯性反射 - 重排序器在线更新
    retrospective_reflection(R_M, g_φ)
    
    # 5. 会话状态更新
    S = update_session_state(S, q, a)
    
    return a, S
```
### 1. 记忆检索节点
```
def memory_retrieval(query, memory_bank, retriever, top_k=10):
    """从记忆库中检索相关记忆"""
    query_embedding = retriever.encode(query)
    similar_memories = memory_bank.search(query_embedding, top_k)
    
    M_K = []
    for memory_id, similarity_score in similar_memories:
        memory_entry = memory_bank.memories[memory_id]
        M_K.append({
            'memory': memory_entry,
            'similarity_score': similarity_score
        })
    
    return M_K
```
### 2. 记忆重排序节点
```
def memory_reranking(query, candidate_memories, reranker, top_m=5):
    """对检索结果进行精细化重排序"""
    query_emb = retriever.encode(query)
    
    # 计算重排序分数
    rerank_scores = []
    for candidate in candidate_memories:
        memory_emb = candidate['memory']['embedding']
        
        # 计算交互特征（如时序信息、共现频率等）
        interaction_features = calculate_interaction_features(query, candidate)
        
        # 重排序得分
        rerank_score = reranker(query_emb, memory_emb, interaction_features)
        rerank_scores.append(rerank_score)
    
    # 按重排序分数排序并选择top-M
    sorted_indices = torch.argsort(torch.tensor(rerank_scores), descending=True)
    M_M = [candidate_memories[i] for i in sorted_indices[:top_m]]
    
    return M_M
```
### 3. 响应生成与归因分析节点
```
def response_generation_with_attribution(query, session_history, memories, generator):
    """生成回复并进行归因分析"""
    # 构建提示
    prompt = build_prompt(query, session_history, memories)
    
    # 生成回复
    response = generator.generate(prompt)
    
    # 归因分析 - LLM判断每个记忆是否被使用
    attribution_prompt = f"""
    回复: {response}
    参考记忆:
    {format_memories_for_attribution(memories)}
    
    请判断回复是否参考了每个记忆，输出JSON格式:
    {{"attributions": [{{"memory_id": 1, "used": true/false, "reason": "..."}}]}}
    """
    
    attribution_result = generator.generate(attribution_prompt)
    R_M = parse_attribution_result(attribution_result)
    
    return response, R_M

def build_prompt(query, session_history, memories):
    """构建生成提示"""
    memory_context = "\n".join([
        f"记忆 {i+1}: {mem['memory']['topic_summary']}\n详情: {mem['memory']['raw_dialogue']}"
        for i, mem in enumerate(memories)
    ])
    
    prompt = f"""
    会话历史:
    {format_session_history(session_history)}
    
    相关记忆:
    {memory_context}
    
    当前查询: {query}
    
    请基于以上信息生成自然、有帮助的回复:
    """
    return prompt
```
### 4. 回溯性反射节点
```
def retrospective_reflection(attributions, reranker, learning_rate=0.001):
    """基于归因反馈更新重排序器"""
    optimizer = torch.optim.Adam(reranker.parameters(), lr=learning_rate)
    
    rewards = []
    for attribution in attributions:
        if attribution['used']:
            rewards.append(1.0)  # 正奖励
        else:
            rewards.append(-0.5)  # 负奖励
    
    # 强化学习更新（简化版）
    if rewards:
        avg_reward = sum(rewards) / len(rewards)
        loss = -torch.log(torch.sigmoid(torch.tensor(avg_reward)))
        
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
```
### 5. 会话状态更新节点
```
def update_session_state(session_history, query, response):
    """更新会话历史"""
    new_turn = {
        'query': query,
        'response': response,
        'timestamp': datetime.now()
    }
    session_history.append(new_turn)
    return session_history
```
## 三、会话结束时的离线处理流程
### 算法2：会话结束处理流程
```
def offline_processing_algorithm(S, B, generator):
    """
    会话结束时的离线处理
    """
    # 1. 会话结束判断
    if not is_session_ended(S):
        return B, S
    
    # 2. 记忆提取
    new_memories = memory_extraction(S, generator)
    
    # 3. 记忆整合
    updated_memory_bank = memory_integration(new_memories, B, generator)
    
    # 4. 会话重置
    S = reset_session()
    
    return updated_memory_bank, S
```
### 1. 会话结束判断节点
```
def is_session_ended(session_history, timeout_minutes=30):
    """判断会话是否结束"""
    if not session_history:
        return False
    
    last_turn = session_history[-1]
    time_since_last = datetime.now() - last_turn['timestamp']
    
    # 超时或明确结束信号
    return (time_since_last.total_seconds() > timeout_minutes * 60 or 
            contains_end_signal(last_turn['query']))
```

### 3. 记忆整合节点
```
def memory_integration(new_memories, memory_bank, generator):
    """将新记忆整合到记忆库中"""
    for new_memory in new_memories:
        # 查找相似记忆
        similar_memories = memory_bank.search(
            retriever.encode(new_memory['topic_summary']), 
            top_k=3
        )
        
        if similar_memories:
            # 判断是否合并
            should_merge = check_memory_similarity(new_memory, similar_memories[0], generator)
            
            if should_merge:
                # 合并记忆
                merged_memory = merge_memories(new_memory, similar_memories[0], generator)
                memory_bank.update_memory(similar_memories[0]['id'], merged_memory)
            else:
                # 添加为新记忆
                memory_bank.add_memory(
                    new_memory['topic_summary'],
                    new_memory['raw_dialogue'],
                    retriever.encode(new_memory['topic_summary'])
                )
        else:
            # 直接添加
            memory_bank.add_memory(
                new_memory['topic_summary'],
                new_memory['raw_dialogue'],
                retriever.encode(new_memory['topic_summary'])
            )
    
    return memory_bank

def check_memory_similarity(new_memory, existing_memory, generator):
    """使用LLM判断两个记忆是否应该合并"""
    similarity_prompt = f"""
    判断以下两个记忆主题是否相关，应该合并:
    
    新记忆: {new_memory['topic_summary']}
    现有记忆: {existing_memory['topic_summary']}
    
    如果主题高度相关（如同一个话题的不同方面），输出true，否则输出false。
    只输出true或false。
    """
    
    result = generator.generate(similarity_prompt)
    return result.strip().lower() == 'true'
```

## 四、核心工作流总结
### 系统架构图
```
┌─────────────────┐    ┌──────────────────┐
│   在线处理循环   │    │   离线处理循环    │
│   (高频执行)     │    │   (会话结束时)    │
│                 │    │                  │
│ 1. 记忆检索     │◄───┤ 1. 记忆提取      │
│ 2. 记忆重排序   │    │ 2. 记忆整合      │
│ 3. 响应生成     │    │ 3. 会话重置      │
│ 4. 回溯性反射   │    │                  │
│ 5. 会话更新     │    │                  │
└─────────┬───────┘    └────────┬─────────┘
          │                      │
          └─────共享记忆库 B ─────┘
```
### 主要特性对比
|处理循环|触发频率|主要功能|技术核心|
|-|-|-|-|
|**在线循环**|每次用户查询|实时响应生成、检索优化|强化学习、实时更新|
|**离线循环**|会话结束时|记忆库维护、长期优化|主题分析、记忆合并|

### 完整工作流示例
```
class RMMSystem:
    def __init__(self, config):
        # 初始化所有组件
        self.memory_bank = MemoryBank(config['vector_db'])
        self.retriever = DenseRetriever()
        self.reranker = Reranker()
        self.generator = DialogueGenerator(config['model_name'])
        self.session_history = []
        
    def process_query(self, user_query):
        """处理单次用户查询"""
        # 在线处理
        response, updated_session = online_processing_algorithm(
            user_query, 
            self.memory_bank,
            self.session_history,
            self.retriever,
            self.reranker,
            self.generator
        )
        
        self.session_history = updated_session
        
        # 检查是否需要离线处理
        if is_session_ended(self.session_history):
            self.memory_bank, self.session_history = offline_processing_algorithm(
                self.session_history,
                self.memory_bank,
                self.generator
            )
        
        return response
```
该系统通过在线循环实现实时自适应，通过离线循环实现长期记忆优化，共同构成了完整的反射式记忆管理系统。
